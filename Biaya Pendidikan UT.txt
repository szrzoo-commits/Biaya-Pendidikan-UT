const ADMIN_FEES = {
  pendaftaran: 100000,
  almamater_etc: 0 // Asumsi dasar simulasi awal
};

const formatRp = (angka) => {
  if (angka === null || angka === undefined) return '-';
  return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', minimumFractionDigits: 0 }).format(angka);
};

  const [isOpen, setIsOpen] = useState(false);
  return (

      <button 
        onClick={() => setIsOpen(!isOpen)}

  );
};

const TuitionCalculator = () => {
  const [selFakultas, setSelFakultas] = useState('');
  const [selProdi, setSelProdi] = useState('');
  const [selLayanan, setSelLayanan] = useState('');
  const [jmlSks, setJmlSks] = useState(18); // Default avg SKS
  const [isNewStudent, setIsNewStudent] = useState(true);

  // Filter prodi based on selected faculty
  const availableProdi = useMemo(() => {
    if (!selFakultas) return [];
    return PRODI_DATA.filter(p => p.faculty === selFakultas);
  }, [selFakultas]);

  const selectedProdiData = useMemo(() => {
    return PRODI_DATA.find(p => p.id.toString() === selProdi);
  }, [selProdi]);

  // Kalkulasi Otomatis
  const calculation = useMemo(() => {
    if (!selectedProdiData || !selLayanan) return null;

    let baseFee = 0;
    let type = '';
    let sksDetail = '';

    if (selLayanan === 'non_sipas') {
      baseFee = selectedProdiData.sksFee * jmlSks;
      type = 'Non-SIPAS';
      sksDetail = `${jmlSks} SKS x ${formatRp(selectedProdiData.sksFee)}`;
    } else {
      baseFee = selectedProdiData[selLayanan];
      type = 'SIPAS (Paket)';
      sksDetail = 'Paket Semester';
    }

    const adminFee = isNewStudent ? ADMIN_FEES.pendaftaran : 0;
    const total = baseFee + adminFee;

    return { baseFee, adminFee, total, type, sksDetail };
  }, [selectedProdiData, selLayanan, jmlSks, isNewStudent]);

  return (
    <div className="bg-white rounded-2xl shadow-xl border border-gray-100 overflow-hidden flex flex-col lg:flex-row" id="kalkulator">
      {/* Kiri: Form Input */}
      <div className="flex-1 p-6 lg:p-10 border-b lg:border-b-0 lg:border-r border-gray-100">
        <div className="flex items-center gap-3 mb-6">
          <div className="w-10 h-10 rounded-full bg-blue-100 flex items-center justify-center text-[#00509E]">
            <Calculator className="w-5 h-5" />
          </div>
          <h3 className="text-2xl font-bold text-gray-900">Kalkulator Simulasi Biaya</h3>
        </div>

        <div className="space-y-5">
          {/* Step 1 */}
          <div>
            <label className="block text-sm font-semibold text-gray-700 mb-2">1. Pilih Fakultas</label>
            <select 
              className="w-full p-3 bg-gray-50 border border-gray-200 rounded-lg focus:ring-2 focus:ring-[#00509E] focus:border-[#00509E] outline-none transition-all"
              value={selFakultas}
              onChange={(e) => {
                setSelFakultas(e.target.value);
                setSelProdi('');
                setSelLayanan('');
              }}
            >
              <option value="">-- Pilih Fakultas --</option>
              {FACULTIES.map(f => (
                <option key={f.id} value={f.id}>{f.name}</option>
              ))}
            </select>
          </div>

          {/* Step 2 */}
          <div>
            <label className="block text-sm font-semibold text-gray-700 mb-2">2. Pilih Program Studi</label>
            <select 
              className="w-full p-3 bg-gray-50 border border-gray-200 rounded-lg focus:ring-2 focus:ring-[#00509E] outline-none transition-all disabled:opacity-50"
              value={selProdi}
              onChange={(e) => {
                setSelProdi(e.target.value);
                setSelLayanan('');
              }}
              disabled={!selFakultas}
            >
              <option value="">-- Pilih Program Studi --</option>
              {availableProdi.map(p => (
                <option key={p.id} value={p.id.toString()}>{p.name}</option>
              ))}
            </select>
          </div>

          {/* Step 3 */}
          <div>
            <label className="block text-sm font-semibold text-gray-700 mb-2">3. Pilih Skema Layanan</label>
            <select 
              className="w-full p-3 bg-gray-50 border border-gray-200 rounded-lg focus:ring-2 focus:ring-[#00509E] outline-none transition-all disabled:opacity-50"
              value={selLayanan}
              onChange={(e) => setSelLayanan(e.target.value)}
              disabled={!selProdi}
            >
              <option value="">-- Pilih Skema --</option>
              {selectedProdiData && LAYANAN_OPTIONS.map(opt => {
                // Mencegah pemilihan SIPAS yang nilainya kosong
                if (opt.id !== 'non_sipas' && selectedProdiData[opt.id] === null) return null;
                // Mencegah pemilihan Non-SIPAS untuk prodi yang tidak mendukung (seperti PGSD/PGPAUD)
                if (opt.id === 'non_sipas' && selectedProdiData.sksFee === null) return null;
                return (
                  <option key={opt.id} value={opt.id}>{opt.name}</option>
                );
              })}
            </select>
          </div>

          {/* Step 4 (Conditional) */}
          {selLayanan === 'non_sipas' && (
            <div className="animate-in fade-in slide-in-from-top-2 duration-300">
              <label className="block text-sm font-semibold text-gray-700 mb-2">4. Jumlah SKS yang Diambil</label>
              <div className="flex items-center gap-3">
                <input 
                  type="range" min="12" max="24" 
                  value={jmlSks} 
                  onChange={(e) => setJmlSks(parseInt(e.target.value))}
                  className="w-full h-2 bg-gray-200 rounded-lg appearance-none cursor-pointer accent-[#00509E]"
                />
                <span className="font-bold text-gray-900 min-w-[3rem] text-right">{jmlSks} SKS</span>
              </div>
            </div>
          )}

          {/* Opsi Tambahan */}
          <div className="pt-4 border-t border-gray-100">
            <label className="flex items-start gap-3 cursor-pointer group">
              <input 
                type="checkbox" 
                checked={isNewStudent}
                onChange={(e) => setIsNewStudent(e.target.checked)}
                className="mt-1 w-4 h-4 text-[#00509E] rounded border-gray-300 focus:ring-[#00509E]"
              />
              <div>
                <span className="block font-semibold text-gray-800 group-hover:text-[#00509E] transition-colors">Saya Mahasiswa Baru (Semester 1)</span>
                <span className="text-xs text-gray-500">Termasuk Biaya Pendaftaran Mahasiswa Baru (Rp 100.000) sesuai Lampiran Tabel 3.6</span>
              </div>
            </label>
          </div>
        </div>
      </div>

      {/* Kanan: Preview Card */}
      <div className="flex-1 bg-gradient-to-br from-[#00509E] to-[#003b7a] p-6 lg:p-10 text-white flex flex-col justify-between">
        <div>
          <h4 className="text-xl font-bold mb-6 flex items-center gap-2">
            <Wallet className="w-6 h-6 text-[#FFD100]" />
            Estimasi Biaya Pendidikan
          </h4>
          
          {!calculation ? (
            <div className="h-48 flex flex-col items-center justify-center text-center opacity-70 border-2 border-dashed border-blue-400 rounded-xl p-6">
              <Info className="w-8 h-8 mb-3 text-blue-200" />
              <p>Lengkapi formulir di sebelah kiri untuk melihat rincian biaya secara otomatis.</p>
            </div>
          ) : (
            <div className="space-y-4 animate-in zoom-in-95 duration-300">
              <div className="bg-white/10 rounded-lg p-4 flex justify-between items-center backdrop-blur-sm">
                <div>
                  <p className="text-blue-200 text-sm mb-1">Uang Kuliah ({calculation.type})</p>
                  <p className="text-xs text-blue-300">{calculation.sksDetail}</p>
                </div>
                <p className="font-bold text-lg">{formatRp(calculation.baseFee)}</p>
              </div>

              {calculation.adminFee > 0 && (
                <div className="bg-white/10 rounded-lg p-4 flex justify-between items-center backdrop-blur-sm">
                  <div>
                    <p className="text-blue-200 text-sm mb-1">Biaya Pendaftaran Maba</p>
                    <p className="text-xs text-blue-300">Dibayarkan 1x di awal</p>
                  </div>
                  <p className="font-bold text-lg">{formatRp(calculation.adminFee)}</p>
                </div>
              )}
            </div>
          )}
        </div>

        <div className="mt-8 pt-6 border-t border-blue-400/30">
          <p className="text-blue-200 text-sm mb-2">Total Biaya (Semester Pertama)</p>
          <div className="text-4xl lg:text-5xl font-extrabold text-[#FFD100] tracking-tight">
            {calculation ? formatRp(calculation.total) : 'Rp 0'}
          </div>
          {calculation && (
             <button className="mt-6 w-full bg-[#FFD100] hover:bg-yellow-500 text-[#00509E] font-bold py-3 px-6 rounded-lg shadow-lg hover:shadow-xl transition-all transform hover:-translate-y-0.5 flex justify-center items-center gap-2">
                Simpan & Unduh Rincian <ChevronRight className="w-5 h-5" />
             </button>
          )}
        </div>
      </div>
    </div>
  );
};
